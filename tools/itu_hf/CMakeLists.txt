cmake_minimum_required(VERSION 3.16)
project(FobosItuHf LANGUAGES C CXX)
if(WIN32)
    if(NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
        message(FATAL_ERROR "The optional Windows HF helper requires x64")
    endif()
elseif(CMAKE_SYSTEM_NAME STREQUAL "Linux")
    if(NOT CMAKE_SIZEOF_VOID_P EQUAL 4 AND NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
        message(FATAL_ERROR "The optional Linux HF helper requires a native 32/64-bit build")
    endif()
else()
    message(FATAL_ERROR "The optional HF helper requires Windows or Linux")
endif()
get_filename_component(HF_ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
set(HF_VENDOR "${HF_ROOT}/third_party/itu_hf")
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
set(CMAKE_LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 17)
add_compile_definitions(_CRT_SECURE_NO_WARNINGS)
set(p372 "${HF_VENDOR}/P372/Src/P372")
add_library(itu_hf_p372 SHARED
    "${p372}/InitializeNoise.c" "${p372}/MakeNoise.c"
    "${p372}/Noise.c" "${p372}/NoiseMemory.c")
set_target_properties(itu_hf_p372 PROPERTIES OUTPUT_NAME P372)
if(NOT WIN32)
    set_target_properties(itu_hf_p372 PROPERTIES OUTPUT_NAME p372)
    target_compile_options(itu_hf_p372 PRIVATE -include errno.h)
    target_link_libraries(itu_hf_p372 PRIVATE m)
endif()
set(p533 "${HF_VENDOR}/P533/Src/P533")
file(GLOB hf_p533_sources CONFIGURE_DEPENDS "${p533}/*.c")
add_library(itu_hf_p533 SHARED ${hf_p533_sources})
set_target_properties(itu_hf_p533 PROPERTIES OUTPUT_NAME P533)
if(NOT WIN32)
    set_target_properties(itu_hf_p533 PROPERTIES OUTPUT_NAME p533)
    target_link_libraries(itu_hf_p533 PRIVATE m ${CMAKE_DL_LIBS})
endif()
add_dependencies(itu_hf_p533 itu_hf_p372)
set(cli "${HF_VENDOR}/ITURHFProp/Src/ITURHFProp")
set_source_files_properties("${cli}/ITURHFProp.c" PROPERTIES COMPILE_DEFINITIONS "main=itu_hf_upstream_main")
add_executable(itu_hf_helper "${HF_ROOT}/tools/itu_hf_helper.c"
    "${cli}/ITURHFProp.c" "${cli}/ReadInputConfiguration.c"
    "${cli}/ValidateITURHFP.c" "${cli}/Report.c" "${cli}/DumpPathData.c")
add_dependencies(itu_hf_helper itu_hf_p533 itu_hf_p372)
if(MSVC)
    foreach(target itu_hf_p372 itu_hf_p533 itu_hf_helper)
        set_property(TARGET ${target} PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
    endforeach()
endif()
if(NOT WIN32)
    # The official C headers define shared globals (pre-GCC-10 common linkage).
    foreach(target itu_hf_p372 itu_hf_p533 itu_hf_helper)
        target_compile_options(${target} PRIVATE -fcommon -Wno-error=date-time)
    endforeach()
    target_link_libraries(itu_hf_helper PRIVATE m ${CMAKE_DL_LIBS})
    set_target_properties(itu_hf_helper PROPERTIES BUILD_RPATH "$ORIGIN" INSTALL_RPATH "$ORIGIN")
endif()
option(ITU_HF_BUILD_QT "Build the isolated Qt dialog and self-tests" ON)
if(ITU_HF_BUILD_QT)
    find_package(Qt5 5.15 REQUIRED COMPONENTS Core Widgets Test)
    add_subdirectory("${HF_ROOT}/src/propagation/ituhf" qt_hf)
    enable_testing()
    add_executable(itu_hf_selftest "${HF_ROOT}/tools/itu_hf_selftest.cpp")
    target_link_libraries(itu_hf_selftest PRIVATE fobos_hf_prediction Qt5::Test)
    set_target_properties(itu_hf_selftest PROPERTIES AUTOMOC ON)
    add_test(NAME itu_hf_selftest COMMAND itu_hf_selftest
        "$<TARGET_FILE:itu_hf_helper>" "${ITU_HF_DATA_DIR}")
    set_tests_properties(itu_hf_selftest PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
endif()
enable_testing()
find_package(Python3 QUIET COMPONENTS Interpreter)
set(ITU_HF_DATA_DIR "" CACHE PATH "Optional model-test data directory")
if(Python3_Interpreter_FOUND AND NOT CMAKE_CROSSCOMPILING)
    add_test(NAME itu_hf_modeltest COMMAND "${Python3_EXECUTABLE}"
        "${HF_ROOT}/tools/itu_hf_modeltest.py" --helper "$<TARGET_FILE:itu_hf_helper>"
        --vendor "${HF_VENDOR}" --data "${ITU_HF_DATA_DIR}")
endif()
install(TARGETS itu_hf_helper itu_hf_p533 itu_hf_p372 RUNTIME DESTINATION itu_hf LIBRARY DESTINATION itu_hf)
install(FILES "${HF_VENDOR}/README.md" "${HF_VENDOR}/VENDOR.md" DESTINATION itu_hf/source-notices)
foreach(component P533 P372 ITURHFProp)
    install(DIRECTORY "${HF_VENDOR}/${component}/Src/" DESTINATION "itu_hf/source/${component}/Src"
        FILES_MATCHING PATTERN "*.c" PATTERN "*.h")
endforeach()
install(FILES "${HF_VENDOR}/SOURCE_MANIFEST.json" DESTINATION itu_hf/source-notices OPTIONAL)
