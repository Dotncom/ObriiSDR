cmake_minimum_required(VERSION 3.16)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(ObriiLoRaWAN LANGUAGES CXX)
    enable_testing()
    set(OBRII_LORAWAN_STANDALONE ON)
else()
    set(OBRII_LORAWAN_STANDALONE OFF)
endif()
option(OBRII_LORAWAN_BUILD_TESTS "Build LoRaWAN self tests" ${OBRII_LORAWAN_STANDALONE})
option(OBRII_LORAWAN_WITH_LORA "Build LoRaWAN IQ adapter" ON)
option(OBRII_LORAWAN_BUILD_TOOLS "Build offline LoRaWAN protocol tool" ${OBRII_LORAWAN_STANDALONE})
add_library(ObriiLoRaWAN STATIC lorawan.h crypto.cpp codec.cpp session.cpp state.cpp region.cpp regional.h version11.h version11.cpp classbc.h classbc.cpp)
target_compile_features(ObriiLoRaWAN PUBLIC cxx_std_17)
target_include_directories(ObriiLoRaWAN PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
set_target_properties(ObriiLoRaWAN PROPERTIES POSITION_INDEPENDENT_CODE ON)
if(MSVC)
    target_compile_options(ObriiLoRaWAN PRIVATE /W4 /utf-8)
else()
    target_compile_options(ObriiLoRaWAN PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(OBRII_LORAWAN_WITH_LORA)
    if(NOT TARGET ObriiLoRa)
        add_subdirectory(../lora ${CMAKE_CURRENT_BINARY_DIR}/lora)
    endif()
    target_sources(ObriiLoRaWAN PRIVATE lorawanlora.cpp lorawanlora.h)
    target_link_libraries(ObriiLoRaWAN PUBLIC ObriiLoRa)
    target_compile_definitions(ObriiLoRaWAN PUBLIC OBRII_LORAWAN_WITH_LORA=1)
    install(FILES lorawanlora.h DESTINATION include/obrii/lorawan)
endif()
if(OBRII_LORAWAN_BUILD_TESTS)
    add_executable(ObriiLoRaWANSelfTest tests/selftest.cpp)
    target_link_libraries(ObriiLoRaWANSelfTest PRIVATE ObriiLoRaWAN)
    target_compile_definitions(ObriiLoRaWANSelfTest PRIVATE OBRII_LORAWAN_VECTORS="${CMAKE_CURRENT_SOURCE_DIR}/tests/vectors.tsv")
    add_test(NAME ObriiLoRaWANSelfTest COMMAND ObriiLoRaWANSelfTest)
    set_tests_properties(ObriiLoRaWANSelfTest PROPERTIES TIMEOUT 90)
    add_executable(ObriiLoRaWANBC11SelfTest tests/selftest_bc11.cpp)
    target_link_libraries(ObriiLoRaWANBC11SelfTest PRIVATE ObriiLoRaWAN)
    target_compile_definitions(ObriiLoRaWANBC11SelfTest PRIVATE OBRII_LORAWAN_VECTORS11="${CMAKE_CURRENT_SOURCE_DIR}/tests/vectors11.tsv")
    add_test(NAME ObriiLoRaWANBC11SelfTest COMMAND ObriiLoRaWANBC11SelfTest)
    set_tests_properties(ObriiLoRaWANBC11SelfTest PROPERTIES TIMEOUT 90)
endif()
if(OBRII_LORAWAN_BUILD_TOOLS)
    add_executable(ObriiLoRaWANTool tool.cpp)
    target_link_libraries(ObriiLoRaWANTool PRIVATE ObriiLoRaWAN)
    install(TARGETS ObriiLoRaWANTool RUNTIME DESTINATION bin)
endif()
install(TARGETS ObriiLoRaWAN ARCHIVE DESTINATION lib)
install(FILES lorawan.h regional.h version11.h classbc.h DESTINATION include/obrii/lorawan)
install(DIRECTORY obrii_lorawan_gateway DESTINATION share/obrii/lorawan FILES_MATCHING PATTERN "*.py")
install(FILES NOTICE ../../third_party/tiny_aes/unlicense.txt DESTINATION share/obrii/lorawan)
