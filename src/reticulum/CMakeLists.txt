cmake_minimum_required(VERSION 3.16)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(ObriiReticulum LANGUAGES CXX)
    enable_testing()
    set(OBRII_RETICULUM_STANDALONE ON)
endif()
option(OBRII_RETICULUM_BUILD_TESTS "Build native Reticulum LoRa bearer tests and offline IQ pipe" ${OBRII_RETICULUM_STANDALONE})
if(NOT TARGET ObriiLoRa)
    add_subdirectory(../lora ${CMAKE_CURRENT_BINARY_DIR}/lora)
endif()
add_library(ObriiReticulumLoRa STATIC lora/reticulumlora.cpp lora/reticulumlora.h)
target_link_libraries(ObriiReticulumLoRa PUBLIC ObriiLoRa)
target_include_directories(ObriiReticulumLoRa PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/lora)
target_compile_features(ObriiReticulumLoRa PUBLIC cxx_std_17)
set_target_properties(ObriiReticulumLoRa PROPERTIES POSITION_INDEPENDENT_CODE ON)
if(MSVC)
    target_compile_options(ObriiReticulumLoRa PRIVATE /W4 /utf-8)
else()
    target_compile_options(ObriiReticulumLoRa PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(OBRII_RETICULUM_BUILD_TESTS)
    add_executable(ObriiReticulumLoRaSelfTest tests/loratest.cpp)
    target_link_libraries(ObriiReticulumLoRaSelfTest PRIVATE ObriiReticulumLoRa)
    add_test(NAME ObriiReticulumLoRaSelfTest COMMAND ObriiReticulumLoRaSelfTest)
    set_tests_properties(ObriiReticulumLoRaSelfTest PROPERTIES TIMEOUT 90)
    add_executable(ObriiReticulumIqPipe tests/iqpipe.cpp)
    target_link_libraries(ObriiReticulumIqPipe PRIVATE ObriiReticulumLoRa)
endif()
install(TARGETS ObriiReticulumLoRa ARCHIVE DESTINATION lib)
install(FILES lora/reticulumlora.h DESTINATION include/obrii/reticulum)
install(DIRECTORY obrii_reticulum DESTINATION share/obrii/reticulum
        FILES_MATCHING PATTERN "*.py")
install(FILES requirements.txt requirements-lock.txt NOTICE LICENSE.Reticulum.txt DESTINATION share/obrii/reticulum)
