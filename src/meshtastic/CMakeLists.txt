cmake_minimum_required(VERSION 3.16)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(ObriiMeshtastic LANGUAGES CXX)
    enable_testing()
    set(OBRII_MESHTASTIC_STANDALONE ON)
else()
    set(OBRII_MESHTASTIC_STANDALONE OFF)
endif()
option(OBRII_MESHTASTIC_BUILD_TESTS "Build Meshtastic core tests" ${OBRII_MESHTASTIC_STANDALONE})
option(OBRII_MESHTASTIC_BUILD_TOOLS "Build offline Meshtastic tool" ${OBRII_MESHTASTIC_STANDALONE})
option(OBRII_MESHTASTIC_WITH_LORA "Build LoRa IQ adapter" ON)
add_library(ObriiMeshtastic STATIC meshprotocol.cpp meshprotobuf.cpp meshnode.cpp aes128.cpp aes256.cpp meshtastic.h meshpki.cpp meshpki.h
    meshadmincodec.cpp meshadminserver.cpp meshadminclient.cpp meshadmin.h)
if(WIN32)
    target_sources(ObriiMeshtastic PRIVATE meshcrypto_cng.cpp)
    target_link_libraries(ObriiMeshtastic PRIVATE bcrypt)
else()
    find_package(OpenSSL 1.1.1 REQUIRED COMPONENTS Crypto)
    target_sources(ObriiMeshtastic PRIVATE meshcrypto_openssl.cpp)
    target_link_libraries(ObriiMeshtastic PRIVATE OpenSSL::Crypto)
endif()
target_compile_features(ObriiMeshtastic PUBLIC cxx_std_17)
target_include_directories(ObriiMeshtastic PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
set_target_properties(ObriiMeshtastic PROPERTIES POSITION_INDEPENDENT_CODE ON)
if(MSVC)
    target_compile_options(ObriiMeshtastic PRIVATE /W4 /utf-8)
else()
    target_compile_options(ObriiMeshtastic PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(OBRII_MESHTASTIC_WITH_LORA)
    if(NOT TARGET ObriiLoRa)
        add_subdirectory(../lora ${CMAKE_CURRENT_BINARY_DIR}/lora)
    endif()
    target_sources(ObriiMeshtastic PRIVATE meshlora.cpp meshlora.h)
    target_link_libraries(ObriiMeshtastic PUBLIC ObriiLoRa)
    target_compile_definitions(ObriiMeshtastic PUBLIC OBRII_MESHTASTIC_WITH_LORA=1)
    install(FILES meshlora.h DESTINATION include/obrii/meshtastic)
endif()
if(OBRII_MESHTASTIC_BUILD_TESTS)
    add_executable(ObriiMeshtasticSelfTest meshtest.cpp)
    target_link_libraries(ObriiMeshtasticSelfTest PRIVATE ObriiMeshtastic)
    target_compile_definitions(ObriiMeshtasticSelfTest PRIVATE OBRII_MESHTASTIC_FIXTURES="${CMAKE_CURRENT_SOURCE_DIR}/vectors.tsv")
    if(MSVC)
        target_compile_options(ObriiMeshtasticSelfTest PRIVATE /utf-8)
    endif()
    add_test(NAME ObriiMeshtasticSelfTest COMMAND ObriiMeshtasticSelfTest)
    set_tests_properties(ObriiMeshtasticSelfTest PROPERTIES TIMEOUT 90)
    add_executable(ObriiMeshtasticPkiTest meshpkitest.cpp)
    target_link_libraries(ObriiMeshtasticPkiTest PRIVATE ObriiMeshtastic)
    target_compile_definitions(ObriiMeshtasticPkiTest PRIVATE OBRII_MESHTASTIC_PKI_FIXTURES="${CMAKE_CURRENT_SOURCE_DIR}/pki_vectors.tsv")
    if(MSVC)
        target_compile_options(ObriiMeshtasticPkiTest PRIVATE /utf-8)
    endif()
    add_test(NAME ObriiMeshtasticPkiTest COMMAND ObriiMeshtasticPkiTest)
    set_tests_properties(ObriiMeshtasticPkiTest PROPERTIES TIMEOUT 90)
endif()
if(OBRII_MESHTASTIC_BUILD_TOOLS)
    add_executable(ObriiMeshtasticTool meshtool.cpp)
    target_link_libraries(ObriiMeshtasticTool PRIVATE ObriiMeshtastic)
    install(TARGETS ObriiMeshtasticTool RUNTIME DESTINATION bin)
endif()
install(TARGETS ObriiMeshtastic ARCHIVE DESTINATION lib)
install(FILES meshtastic.h meshpki.h meshadmin.h DESTINATION include/obrii/meshtastic)
install(FILES NOTICE ../../third_party/tiny_aes/unlicense.txt DESTINATION share/obrii/meshtastic)
