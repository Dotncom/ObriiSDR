cmake_minimum_required(VERSION 3.16)
# May be configured on its own while the main application's CMake is being edited.
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(ObriiMeshDesigner VERSION 0.1.0 LANGUAGES CXX)
    enable_testing()
    set(CMAKE_CXX_STANDARD 17)
    set(CMAKE_CXX_STANDARD_REQUIRED ON)
    if(WIN32 AND NOT Qt5_DIR)
        set(Qt5_DIR "C:/Qt/5.15.2/msvc2019_64/lib/cmake/Qt5" CACHE PATH "Qt SDK")
    endif()
    find_package(Qt5 REQUIRED COMPONENTS Core Widgets Network)
endif()
set(CMAKE_AUTOMOC ON)
add_library(ObriiMeshCore STATIC meshproject.cpp meshproject.h meshruntime.cpp meshruntime.h)
target_link_libraries(ObriiMeshCore PUBLIC Qt5::Core Qt5::Network)
target_include_directories(ObriiMeshCore PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
add_library(ObriiMeshUi STATIC meshwindow.cpp meshwindow.h)
target_link_libraries(ObriiMeshUi PUBLIC ObriiMeshCore Qt5::Widgets)
add_executable(ObriiMeshDesigner WIN32 main.cpp)
target_link_libraries(ObriiMeshDesigner PRIVATE ObriiMeshUi)
add_executable(ObriiMeshSelfTest meshselftest.cpp)
target_link_libraries(ObriiMeshSelfTest PRIVATE ObriiMeshUi)
foreach(target ObriiMeshCore ObriiMeshUi ObriiMeshDesigner ObriiMeshSelfTest)
    if(MSVC)
        target_compile_options(${target} PRIVATE /utf-8)
    endif()
endforeach()
add_test(NAME ObriiMeshSelfTest COMMAND ObriiMeshSelfTest)
set_tests_properties(ObriiMeshSelfTest PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen" TIMEOUT 90)
install(TARGETS ObriiMeshDesigner RUNTIME DESTINATION bin)
