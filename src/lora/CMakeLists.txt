cmake_minimum_required(VERSION 3.16)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(ObriiLoRa LANGUAGES CXX)
    enable_testing()
    set(OBRII_LORA_STANDALONE ON)
else()
    set(OBRII_LORA_STANDALONE OFF)
endif()
option(OBRII_LORA_BUILD_TESTS "Build LoRa tests" ${OBRII_LORA_STANDALONE})
option(OBRII_LORA_BUILD_TOOLS "Build offline LoRa IQ tool" ${OBRII_LORA_STANDALONE})
add_library(ObriiLoRa STATIC loracodec.cpp loramodem.cpp lorareceiver.cpp loraphy.h)
set_target_properties(ObriiLoRa PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_compile_features(ObriiLoRa PUBLIC cxx_std_17)
target_include_directories(ObriiLoRa PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
if(MSVC)
    target_compile_options(ObriiLoRa PRIVATE /W4 /utf-8)
else()
    target_compile_options(ObriiLoRa PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(OBRII_LORA_BUILD_TESTS)
    add_executable(ObriiLoRaSelfTest loratest.cpp)
    target_link_libraries(ObriiLoRaSelfTest PRIVATE ObriiLoRa)
    add_test(NAME ObriiLoRaSelfTest COMMAND ObriiLoRaSelfTest)
    set_tests_properties(ObriiLoRaSelfTest PROPERTIES TIMEOUT 90)
endif()
if(OBRII_LORA_BUILD_TOOLS)
    add_executable(ObriiLoRaTool loratool.cpp)
    target_link_libraries(ObriiLoRaTool PRIVATE ObriiLoRa)
    install(TARGETS ObriiLoRaTool RUNTIME DESTINATION bin)
endif()
install(TARGETS ObriiLoRa ARCHIVE DESTINATION lib)
install(FILES loraphy.h DESTINATION include/obrii/lora)
install(FILES NOTICE LICENSE.LoRaPHY DESTINATION share/obrii/lora)
